        const consoleEl = document.getElementById('console');
        function log(msg, type = 'info') {
            const entry = document.createElement('div');
            entry.className = `log-entry log-${type}`;
            const time = new Date().toISOString().substring(11, 19);

        class PQCSecurityModule {
            constructor() {
                this.hardwareSerial = null;
                this.sessionKey = null;
            }

            async initialize() {
                log('Initializing Hardware Side-Channel Shields & Anti-Cloning Identity...', 'sec');
                this.hardwareSerial = await this.generateHardwareFingerprint();
                document.getElementById('disp-serial').innerText = this.hardwareSerial.substring(0, 8) + '...';
                await this.derivePQCKey();
                this.startEMNoiseMasking();
            }

            async generateHardwareFingerprint() {
                const c = document.createElement('canvas');
                const ctx = c.getContext('2d');
                ctx.textBaseline = "alphabetic";
                ctx.fillStyle = "#f60";
                ctx.fillRect(125,1,62,20);
                ctx.fillStyle = "#069";
                ctx.fillText("TernaryLoom_CuGa_Core", 2, 15);
                const dataUrl = c.toDataURL();
                
                const enc = new TextEncoder();
                const hashBuffer = await crypto.subtle.digest('SHA-256', enc.encode(dataUrl));
                const hashArray = Array.from(new Uint8Array(hashBuffer));
                return hashArray.map(b => b.toString(16).padStart(2, '0')).join('');
            }

            async derivePQCKey() {
                const keyData = window.crypto.getRandomValues(new Uint8Array(32));
                this.sessionKey = await window.crypto.subtle.importKey(
                    "raw", keyData, { name: "AES-GCM", length: 256 }, false, ["encrypt", "decrypt"]
                );
                log('Post-Quantum Lattice Session Key Derived (Kyber-1024 Equivalent)', 'sec');
            }

            startEMNoiseMasking() {
                setInterval(() => {
                    const dummyHeap = new Float64Array(1024);
                    for(let i=0; i<dummyHeap.length; i++) {
                        dummyHeap[i] = Math.sin(i * Math.random());
                    }
                }, 137);
            }
        }

        class TryteEngine {
            constructor() {
                this.states = 243;
            }

            compress3DCoordinates(points) {
                const tryteBuffer = new Uint8Array(points.length);
                for(let i = 0; i < points.length; i++) {
                    const normalized = Math.max(-1, Math.min(1, points[i]));
                    const tryteVal = Math.floor(((normalized + 1) / 2) * 242);
                    tryteBuffer[i] = tryteVal;
                }
                return tryteBuffer;
            }
        }

        class MetamaterialAudioSynth {
            constructor() {
                this.ctx = null;
                this.osc = null;
                this.gain = null;
                this.active = false;
            }

            init() {
                const AudioCtx = window.AudioContext || window.webkitAudioContext;
                this.ctx = new AudioCtx();
                this.osc = this.ctx.createOscillator();
                this.gain = this.ctx.createGain();

                this.osc.type = 'sawtooth';
                this.osc.frequency.setValueAtTime(220, this.ctx.currentTime);
                this.gain.gain.setValueAtTime(0.05, this.ctx.currentTime);

                this.osc.connect(this.gain);
                this.gain.connect(this.ctx.destination);
                this.osc.start();
                this.active = true;
                log('Audio Sonification Engine Online (220 Hz Base carrier)', 'info');
            }

            update(frequency, effervescence) {
                if(!this.active || !this.ctx) return;
                const safeFreq = Math.max(60, Math.min(1200, frequency));
                this.osc.frequency.setTargetAtTime(safeFreq, this.ctx.currentTime, 0.05);
                const safeGain = 0.02 + (effervescence * 0.08);
                this.gain.gain.setTargetAtTime(safeGain, this.ctx.currentTime, 0.05);
            }
        }

        function renderTernaryLoom(t) {
            ctx.fillStyle = "rgba(5, 7, 10, 0.2)";
            ctx.fillRect(0, 0, width, height);

            const centerX = width / 2;
            const centerY = height / 2;
            const numPoints = 600;
            const floatData = new Float32Array(numPoints);

            const effervescenceFactor = Math.abs(Math.sin(t * 0.002 * eta_visc));
            const volumetricExpansion = 1.0 + (effervescenceFactor * 0.35);

            document.getElementById('stat-eff').innerText = (effervescenceFactor * 100).toFixed(2) + ' %';
            document.getElementById('stat-vol').innerText = volumetricExpansion.toFixed(3) + 'x';

            let baseFreq = 220 + (effervescenceFactor * 440);
            document.getElementById('stat-freq').innerText = baseFreq.toFixed(1) + ' Hz';
            audio.update(baseFreq, effervescenceFactor);

            ctx.lineWidth = 1.5;

            for (let i = 0; i < numPoints; i++) {
                const u = (i / numPoints) * Math.PI * 4;
                const v = u * k_wind;

                const r_core = r1 * volumetricExpansion;
                const x = (r_core + 40 * Math.cos(v)) * Math.cos(u);
                const y = (r_core + 40 * Math.cos(v)) * Math.sin(u);
                const z = 40 * Math.sin(v) + 20 * Math.sin(u * k_wind);

                const scale = 300 / (300 + z);
                const px = centerX + x * scale;
                const py = centerY + y * scale;

                floatData[i] = Math.sin(u);

                if (i % 3 === 0) {
                    ctx.strokeStyle = `rgba(6, 182, 212, ${0.4 + scale * 0.4})`;
                } else if (i % 3 === 1) {
                    ctx.strokeStyle = `rgba(217, 70, 239, ${0.4 + scale * 0.4})`;
                } else {
                    ctx.strokeStyle = `rgba(245, 158, 11, ${0.4 + scale * 0.4})`;
                }

                ctx.beginPath();
                ctx.arc(px, py, 1.5 * scale, 0, Math.PI * 2);
                ctx.stroke();
            }

            requestAnimationFrame(() => renderTernaryLoom(performance.now()));
        }
